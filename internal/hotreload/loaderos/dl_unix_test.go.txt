//go:build darwin || linux

package libdl

import (
	"os"
	"runtime"
	"testing"

	"github.com/ebitengine/purego"
)

func TestProcessAndLibraryLookup(t *testing.T) {
	process, err := Open("")
	if err != nil || process == 0 {
		t.Fatalf("dlopen(NULL): handle %x, %v", process, err)
	}
	defer purego.Dlclose(process)
	check := func(handle uintptr) {
		t.Helper()
		address, err := LookupSymbol(handle, "getpid")
		if err != nil || address == 0 {
			t.Fatalf("getpid lookup: %x, %v", address, err)
		}
		var getpid func() int32
		purego.RegisterFunc(&getpid, address)
		if got := getpid(); int(got) != os.Getpid() {
			t.Fatalf("foreign getpid = %d, OS process = %d", got, os.Getpid())
		}
		if address, err := LookupSymbol(handle, "skgo_fixture_missing_symbol"); address != 0 || err == nil {
			t.Fatalf("missing symbol: %x, %v", address, err)
		}
	}
	check(process)
	name := "libc.so.6"
	if runtime.GOOS == "darwin" {
		name = "/usr/lib/libSystem.B.dylib"
	}
	library, err := Open(name)
	if err != nil || library == 0 {
		t.Fatalf("library %s: %x, %v", name, library, err)
	}
	defer purego.Dlclose(library)
	check(library)
	if handle, err := Open("/skgo_fixture/no-such-library"); handle != 0 || err == nil {
		t.Fatalf("missing library: %x, %v", handle, err)
	}
}
