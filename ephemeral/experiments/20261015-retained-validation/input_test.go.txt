// Expected rejection is independently anchored in a changed dependency input,
// rather than inferred from the retained loader's own list of known files.
func TestRetainedCapturedDependencyInputs(t *testing.T) {
 for _, change := range []string{"constant", "type", "import", "new-go-file", "embed-resource"} {
  t.Run(change,func(t *testing.T){
   root:=t.TempDir()
   write:=func(name,body string){t.Helper(); p:=filepath.Join(root,name);if e:=os.MkdirAll(filepath.Dir(p),0755);e!=nil{t.Fatal(e)};if e:=os.WriteFile(p,[]byte(body),0644);e!=nil{t.Fatal(e)}}
   write("go.mod","module example.com/retainedcheck\n\ngo 1.27\n")
   write("target/value.go","package target;import \"example.com/retainedcheck/dependency\";func Value()int{return dependency.Offset}")
   dependency:="package dependency;const Offset=731;type Label int\n"
   if change=="embed-resource" {dependency="package dependency;import _ \"embed\";const Offset=731;type Label int\n//go:embed value.txt\nvar Embedded string\n";write("dependency/value.txt","EMBEDDED_INITIAL_449")}
   write("dependency/value.go",dependency)
   q:=QualificationUse(t,"whole-retained")
   config:=&packages.Config{Dir:root,Env:append(os.Environ(),"GOWORK=off"),Mode:packages.NeedName|packages.NeedFiles|packages.NeedCompiledGoFiles|packages.NeedImports|packages.NeedDeps|packages.NeedTypes|packages.NeedSyntax|packages.NeedTypesInfo|packages.NeedModule}
   if _,e:=q.Load(config,"example.com/retainedcheck/target","./...");e!=nil{t.Fatal("warm baseline:",e)}
   switch change {
   case "constant":write("dependency/value.go","package dependency;const Offset=887;type Label int\n")
   case "type":write("dependency/value.go","package dependency;const Offset=731;type Label string\n")
   case "import":write("dependency/value.go","package dependency;import \"strings\";const Offset=731;type Label int;var Extra=strings.ToUpper(\"extra\")\n")
   case "new-go-file":write("dependency/added.go","package dependency;var AddedInitialization=983\n")
   case "embed-resource":write("dependency/value.txt","EMBEDDED_CHANGED_601")
   }
   _,e:=q.Load(config,"example.com/retainedcheck/target","./...")
   var rebuild *CompleteBuild
   if !errors.As(e,&rebuild){t.Fatalf("changed %s dependency input admitted; want CompleteBuild before using old host/archive; got %v",change,e)}
   t.Logf("changed %s rejected: %v",change,e)
  })
 }
}

func TestRetainedExternalReplacementCaller(t *testing.T) {
 root:=t.TempDir(); app:=filepath.Join(root,"app"); external:=filepath.Join(root,"external")
 write:=func(base,name,body string){t.Helper();p:=filepath.Join(base,name);if e:=os.MkdirAll(filepath.Dir(p),0755);e!=nil{t.Fatal(e)};if e:=os.WriteFile(p,[]byte(body),0644);e!=nil{t.Fatal(e)}}
 write(app,"go.mod","module example.com/retainedcheck\n\ngo 1.27\nrequire example.com/external v0.0.0\nreplace example.com/external => ../external\n")
 write(external,"go.mod","module example.com/external\n\ngo 1.27\n")
 source:=[]byte("package target;type Order int;func Value()int{return 731}\n")
 write(app,"target/value.go",string(source));write(app,"cmd/main.go","package main;import \"example.com/external\";func main(){_=external.Total()}\n")
 write(external,"value.go","package external;import \"example.com/retainedcheck/target\";var _ target.Order;func Total()int{return 449}\n")
 q:=QualificationUse(t,"whole-retained")
 config:=&packages.Config{Dir:app,Env:append(os.Environ(),"GOWORK=off"),Mode:packages.NeedName|packages.NeedFiles|packages.NeedCompiledGoFiles|packages.NeedImports|packages.NeedDeps|packages.NeedTypes|packages.NeedSyntax|packages.NeedTypesInfo|packages.NeedModule}
 before:=map[string][]byte{filepath.Join(app,"target/value.go"):source};after:=map[string][]byte{filepath.Join(app,"target/value.go"):[]byte("package target;type Order int;func Value()int{return 887}\n")}
 classify:=func(pkgs []*packages.Package)error {var target *packages.Package;var all []*packages.Package;packages.Visit(pkgs,nil,func(p *packages.Package){all=append(all,p);if p.PkgPath=="example.com/retainedcheck/target"{target=p}});if target==nil {t.Fatal("target missing")};return Classify(before,after,[]*packages.Package{target},all)}
 roots,e:=q.Load(config,"example.com/retainedcheck/target","./...");if e!=nil{t.Fatal(e)};if e=classify(roots);e!=nil{t.Fatal("initial type-only caller should be safe:",e)}
 write(external,"value.go","package external;import \"example.com/retainedcheck/target\";var _ target.Order;func Total()int{return target.Value()}\n")
 roots,e=q.Load(config,"example.com/retainedcheck/target","./...");if e!=nil{t.Fatal(e)}
 e=classify(roots);var rebuild *CompleteBuild;if !errors.As(e,&rebuild){t.Fatalf("external replacement caller body admitted: %v",e)}
 t.Log("fresh external replacement caller refused:",e)
 // Deliberate missing-caller control: retaining only the target's import graph
 // loses its reverse caller and incorrectly admits the exact same captured edit.
 var target *packages.Package;for _,p:=range roots{if p.PkgPath=="example.com/retainedcheck/target"{target=p}}
 if e:=classify([]*packages.Package{target});e!=nil{t.Fatalf("root-only mutation did not expose missing reverse caller: %v",e)}
 t.Log("BROKEN CONTROL root-only application graph incorrectly admits the caller edit; complete graph is load-bearing")
}
