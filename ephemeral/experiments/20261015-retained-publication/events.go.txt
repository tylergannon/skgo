//go:build skgo_retained_qualification

package hotreload

import (
	"context"
	"encoding/json"
	"fmt"
	"github.com/coder/websocket"
	"github.com/tylergannon/skgo/internal/gen"
	"io"
	"net/http"
	"os"
	"os/exec"
	"path/filepath"
	"sync"
	"syscall"
	"testing"
	"time"
)

func publicationEvents(t *testing.T, ctx context.Context, web string) (func(gen.Revision) error, func([]gen.Revision)) {
	t.Helper()
	config := `export default {server:{host:'127.0.0.1',port:56611,strictPort:true},plugins:[{name:'publication-events',configureServer(server){server.middlewares.use('/__publication',(req,res)=>{const rev=Number(new URL(req.url,'http://localhost').searchParams.get('revision'));if(rev)server.hot.send({type:'custom',event:'publication:ready',data:{revision:rev}});res.end('ok')});}}]}`
	if e := os.WriteFile(filepath.Join(web, "vite.config.ts"), []byte(config), 0644); e != nil {
		t.Fatal(e)
	}
	log, e := os.Create(filepath.Join(web, "publication-vite.log"))
	if e != nil {
		t.Fatal(e)
	}
	v := exec.CommandContext(ctx, "vp", "dev")
	v.Dir = web
	v.Stdout = log
	v.Stderr = log
	v.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	if e = v.Start(); e != nil {
		t.Fatal(e)
	}
	done := make(chan error, 1)
	go func() { done <- v.Wait() }()
	t.Cleanup(func() {
		syscall.Kill(-v.Process.Pid, syscall.SIGKILL)
		<-done
		log.Close()
		raw, _ := os.ReadFile(filepath.Join(web, "publication-vite.log"))
		root, _ := filepath.Abs("../..")
		os.WriteFile(filepath.Join(root, "ephemeral/experiments/20261015-retained-publication/publication-vite-last.log"), raw, 0644)
	})
	client := http.Client{Timeout: 5 * time.Second}
	for end := time.Now().Add(20 * time.Second); ; {
		r, e := client.Get("http://127.0.0.1:56611/__publication")
		if e == nil {
			r.Body.Close()
			break
		}
		if time.Now().After(end) {
			t.Fatal(e)
		}
		time.Sleep(50 * time.Millisecond)
	}
	c, _, e := websocket.Dial(ctx, "ws://127.0.0.1:56611/", &websocket.DialOptions{Subprotocols: []string{"vite-hmr"}})
	if e != nil {
		t.Fatal(e)
	}
	t.Cleanup(func() { c.CloseNow() })
	var mu sync.Mutex
	var seen []gen.Revision
	messages := make(chan gen.Revision, 32)
	go func() {
		for {
			_, b, e := c.Read(ctx)
			if e != nil {
				return
			}
			var m struct {
				Type, Event string
				Data        struct{ Revision gen.Revision }
			}
			if json.Unmarshal(b, &m) == nil && m.Type == "custom" && m.Event == "publication:ready" {
				mu.Lock()
				seen = append(seen, m.Data.Revision)
				mu.Unlock()
				messages <- m.Data.Revision
			}
		}
	}()
	emit := func(revision gen.Revision) error {
		r, e := client.Get(fmt.Sprintf("http://127.0.0.1:56611/__publication?revision=%d", revision))
		if e != nil {
			return e
		}
		io.Copy(io.Discard, r.Body)
		r.Body.Close()
		if r.StatusCode != 200 {
			return fmt.Errorf("event HTTP%d", r.StatusCode)
		}
		select {
		case got := <-messages:
			if got != revision {
				return fmt.Errorf("Vite event wanted%d got%d", revision, got)
			}
		case <-time.After(5 * time.Second):
			return fmt.Errorf("Vite event missing revision%d", revision)
		}
		return nil
	}
	check := func(want []gen.Revision) {
		mu.Lock()
		defer mu.Unlock()
		if fmt.Sprint(seen) != fmt.Sprint(want) {
			t.Fatalf("Vite actual event revisions wanted%v got%v", want, seen)
		}
		t.Logf("ACTUAL_VITE_WEBSOCKET revisions=%v (obsolete2 absent)", seen)
	}
	return emit, check
}
