func qualificationPublication(t *testing.T, ctx context.Context, tc Toolchain, app, work, pkgPath string, before map[string][]byte, cfg generate.Config, source string, process *contractProcess) {
	emit, checkEvents := publicationEvents(t, ctx, filepath.Join(app, "web"))
	project, e := gen.OpenProject(cfg)
	if e != nil {
		t.Fatal(e)
	}
	defer project.Close()
	q := QualificationUse(t, "whole-retained")
	canonical := func(build *gen.ProjectBuild) map[string][]byte {
		out := map[string][]byte{}
		for path, body := range build.Inputs.Files() {
			logical := build.Inputs.AuthoredPath(path)
			if rel, e := filepath.Rel(build.Inputs.Dir, path); e == nil && rel != ".." && !strings.HasPrefix(rel, "../") {
				logical = filepath.Join(app, rel)
			}
			out[logical] = []byte(build.Inputs.AuthoredText(string(body)))
		}
		return out
	}
	compileLease := func(b *gen.ProjectBuild, name string) Candidate {
		t.Helper()
		start := time.Now()
		capture := canonical(b)
		after := map[string][]byte{}
		for path := range before {
			body, ok := capture[path]
			if !ok {
				t.Fatalf("captured compiler input absent%s", path)
			}
			after[path] = body
		}
		q.frozen = capture
		c, e := tc.CompileApplication(ctx, app, pkgPath, filepath.Join(app, "importcfg"), filepath.Join(work, name), before, after)
		if e != nil {
			b.Cancel()
			t.Fatal(e)
		}
		t.Logf("LEASE_NATIVE_COMPILED revision=%d compile=%s files=%d current=%v", b.Revision, time.Since(start), len(capture), b.Current())
		return c
	}
	prepare := func(name string) (*gen.ProjectBuild, Candidate) {
		t.Helper()
		start := time.Now()
		b, e := project.Prepare(ctx)
		if e != nil {
			t.Fatal(e)
		}
		t.Logf("PROJECT_PREPARE revision=%d elapsed=%s", b.Revision, time.Since(start))
		return b, compileLease(b, name)
	}

	writeRevision := func(n int, marker string) {
		t.Helper()
		a := filepath.Join(app, "web/src/routes/orders/orders.remote.go")
		b := filepath.Join(app, "web/src/routes/orders/server.go")
		if e := os.WriteFile(a, []byte(strings.Replace(source, "return total", fmt.Sprintf("return total*0+%d", n), 1)), 0644); e != nil {
			t.Fatal(e)
		}
		if e := os.WriteFile(b, []byte(strings.Replace(contractEndpoint, "endpoint:", marker+":", 1)), 0644); e != nil {
			t.Fatal(e)
		}
		revision, changed, e := project.Observe(ctx, a, b)
		if e != nil || !changed {
			t.Fatalf("observe revision %d changed=%v error=%v", revision, changed, e)
		}
		t.Logf("OBSERVED revision=%d literal=%d marker=%s", revision, n, marker)
	}
	var installed, events []gen.Revision
	admit := func(b *gen.ProjectBuild, c Candidate) error {
		return b.Commit(func(*gen.CompilerInputs) error {
			process.mustCall(t, contractCommand{Op: "install", Candidate: c})
			installed = append(installed, b.Revision)
			events = append(events, b.Revision)
			return emit(b.Revision)
		})
	}
	check := func(n int, marker string) {
		t.Helper()
		for path, want := range map[string]string{"/orders": fmt.Sprintf("<h1>Order total %d</h1>", n), "/orders?raw=1": fmt.Sprintf("%s:%d", marker, n)} {
			headers := map[string]string{"Accept": "text/html"}
			if strings.Contains(path, "raw") {
				headers["Accept"] = "application/json"
			}
			got := process.mustCall(t, contractCommand{Op: "request", Method: "GET", Path: path, Headers: headers})
			if got.Status != 200 || !strings.Contains(got.Body, want) {
				t.Fatalf("literal coherent request path=%s want=%s got=%+v", path, want, got)
			}
			t.Logf("HTTP literal=%s pid=%d", want, got.PID)
		}
	}
	initial, c0 := prepare("initial")
	if e := admit(initial, c0); e != nil {
		t.Fatal(e)
	}
	initial.Inputs.Close()
	check(3000, "endpoint")
	writeRevision(4101, "old4101")
	old, e := project.Prepare(ctx)
	if e != nil {
		t.Fatal(e)
	}
	// Preparation is held after the actual native compiler, across a newer desired
	// multi-file revision. No disk read can retarget this compiler-input lease.
	writeRevision(5202, "current5202")
	cOld := compileLease(old, "obsolete-held-during-newer-disk")
	obj, e := os.ReadFile(cOld.Object)
	if e != nil || !strings.Contains(string(obj), "old4101:") || strings.Contains(string(obj), "current5202:") {
		t.Fatalf("obsolete capture mixed newer endpoint bytes error=%v", e)
	}
	t.Log("IMMUTABLE_MULTIFILE_CAPTURE old native object contains old4101, not current5202, despite newer physical+observed source")
	current, cCurrent := prepare("current")
	if old.Diagnose(errors.New("OBSOLETE_DIAGNOSTIC_4101")) {
		t.Fatal("obsolete diagnostic published")
	}
	oldInstallCalled := false
	e = old.Commit(func(*gen.CompilerInputs) error {
		oldInstallCalled = true
		process.mustCall(t, contractCommand{Op: "install", Candidate: cOld})
		events = append(events, old.Revision)
		return nil
	})
	if !errors.Is(e, gen.ErrObsolete) || oldInstallCalled {
		t.Fatalf("obsolete native install admitted err=%v callback=%v", e, oldInstallCalled)
	}
	old.Cancel()
	check(3000, "endpoint")
	if e := admit(current, cCurrent); e != nil {
		t.Fatal(e)
	}
	current.Inputs.Close()
	check(5202, "current5202")
	if len(installed) != 2 || len(events) != 2 || installed[0] != 1 || installed[1] != 3 || events[0] != 1 || events[1] != 3 {
		t.Fatalf("obsolete publication installed=%v events=%v", installed, events)
	}
	if project.Diagnostic() != nil {
		t.Fatalf("stale diagnostic visible %+v", project.Diagnostic())
	}
	t.Logf("OBSOLETE_CONTROL install/event revisions=%v; stale revision2 rejected before host callback", installed)
	// Current Commit's install/event callback holds the Project lock. Observe
	// cannot announce a later desired revision halfway through publication.
	writeRevision(6303, "linear6303")
	linear, cLinear := prepare("linear")
	entered, release, committed, observed := make(chan struct{}), make(chan struct{}), make(chan error, 1), make(chan struct{})
	go func() {
		committed <- linear.Commit(func(*gen.CompilerInputs) error {
			close(entered)
			<-release
			process.mustCall(t, contractCommand{Op: "install", Candidate: cLinear})
			installed = append(installed, linear.Revision)
			events = append(events, linear.Revision)
			return emit(linear.Revision)
		})
	}()
	<-entered
	nextFile := filepath.Join(app, "web/src/routes/orders/orders.remote.go")
	os.WriteFile(nextFile, []byte(strings.Replace(source, "return total", "return total*0+7407", 1)), 0644)
	go func() {
		_, _, e := project.Observe(ctx, nextFile)
		if e != nil {
			t.Error(e)
		}
		close(observed)
	}()
	select {
	case <-observed:
		t.Fatal("Observe crossed active native publication")
	case <-time.After(30 * time.Millisecond):
	}
	close(release)
	if e := <-committed; e != nil {
		t.Fatal(e)
	}
	<-observed
	linear.Inputs.Close()
	check(6303, "linear6303")
	repaired, cRepair := prepare("later-current")
	if e := admit(repaired, cRepair); e != nil {
		t.Fatal(e)
	}
	repaired.Inputs.Close()
	check(7407, "linear6303")
	t.Logf("LINEARIZED native publication then Observe; installed/events=%v", installed)
	checkEvents([]gen.Revision{1, 3, 4, 5})
	publicationRebuild(t, ctx, tc, app, work, pkgPath, before, source, project, q, canonical, process, emit, checkEvents)
}

func publicationRebuild(t *testing.T, ctx context.Context, tc Toolchain, app, work, pkgPath string, before map[string][]byte, source string, project *gen.Project, q *qualificationLoader, canonical func(*gen.ProjectBuild) map[string][]byte, old *contractProcess, emit func(gen.Revision) error, checkEvents func([]gen.Revision)) {
	t.Helper()
	authored := filepath.Join(app, "web/src/routes/orders/orders.remote.go")
	structural := strings.Replace(source, "quantity:2", "quantity:7", 1)
	if e := os.WriteFile(authored, []byte(structural), 0644); e != nil {
		t.Fatal(e)
	}
	if _, _, e := project.Observe(ctx, authored); e != nil {
		t.Fatal(e)
	}
	build, e := project.Prepare(ctx)
	if e != nil {
		t.Fatal(e)
	}
	defer build.Inputs.Close()
	capture := canonical(build)
	after := map[string][]byte{}
	for path := range before {
		after[path] = capture[path]
	}
	q.frozen = capture
	_, e = tc.CompileApplication(ctx, app, pkgPath, filepath.Join(app, "importcfg"), filepath.Join(work, "incompatible-initializer"), before, after)
	var full *CompleteBuild
	if !errors.As(e, &full) {
		t.Fatalf("incompatible initializer did not choose rebuild: %v", e)
	}
	t.Logf("REBUILD_SELECTED revision=%d reason=%s", build.Revision, e)
	started := time.Now()
	dir := build.Inputs.Dir
	relative := strings.TrimPrefix(pkgPath, "example.com/nativefixture/")
	linkDir := filepath.Join(dir, filepath.FromSlash(relative))
	run := func(tool Toolchain, args ...string) []byte {
		t.Helper()
		raw, e := tool.command(ctx, dir, args...).CombinedOutput()
		if e != nil {
			t.Fatalf("captured rebuild go%v: %v\n%s", args, e, raw)
		}
		return raw
	}
	pp, e := packages.Load(&packages.Config{Context: ctx, Dir: dir, Env: tc.command(ctx, dir).Env, Mode: packages.NeedName | packages.NeedFiles | packages.NeedCompiledGoFiles | packages.NeedImports | packages.NeedDeps | packages.NeedTypes | packages.NeedSyntax | packages.NeedTypesInfo | packages.NeedModule}, pkgPath)
	if e != nil || len(pp) != 1 {
		t.Fatalf("rebuild load %v %v", pp, e)
	}
	newTC := tc
	newTC.AuthoredPaths = map[string]string{}
	newBefore := map[string][]byte{}
	for _, path := range pp[0].CompiledGoFiles {
		body, e := os.ReadFile(path)
		if e != nil {
			t.Fatal(e)
		}
		newBefore[path] = body
		if filepath.Base(path) != "skgo_gen.go" {
			newTC.AuthoredPaths[path] = build.Inputs.AuthoredPath(path)
		}
	}
	lowered, e := LowerAuthored(pp[0], newBefore, newTC.AuthoredPaths)
	if e != nil {
		t.Fatal(e)
	}
	overlays := map[string][]byte{}
	for name, body := range lowered.Host {
		overlays[filepath.Join(linkDir, name)] = body
	}
	main, e := os.ReadFile(filepath.Join(dir, "cmd/main.go"))
	if e != nil {
		t.Fatal(e)
	}
	mainSource := string(main)
	mainSource = strings.Replace(mainSource, "// HOT_INIT", `if err:=orders.SkgoHotInit(16);err!=nil{panic(err)}; install=func(c candidate)error{return orders.SkgoHotInstall(c.Object,c.Storage,c.Reclaim)}; usage=func()(int,int){return orders.SkgoHotUsage()}`, 1)
	mainSource = strings.Replace(mainSource, "// HOT_FETCH", `fetchConfig.CarryContext=orders.SkgoHotContext`, 1)
	mainSource = strings.Replace(mainSource, "// HOT_HANDLER", `handler=orders.SkgoHotHandler(handler)`, 1)
	overlays[filepath.Join(dir, "cmd/main.go")] = []byte(mainSource)
	cfgPath := filepath.Join(work, "rebuilt-importcfg")
	if e = os.WriteFile(cfgPath, run(newTC, "list", "-export", "-deps", "-f", "{{if .Export}}packagefile {{.ImportPath}}={{.Export}}{{end}}", pkgPath), 0644); e != nil {
		t.Fatal(e)
	}
	hostTC, e := newTC.HostView(ctx, dir, filepath.Join(work, "rebuilt-hostview"), overlays, cfgPath)
	if e != nil {
		t.Fatal(e)
	}
	run(hostTC, "mod", "tidy")
	bin := filepath.Join(work, "rebuilt-host")
	run(hostTC, "build", "-ldflags=-s=false", "-o", bin, "./cmd")
	// The frontend build is unchanged by this Go initializer change. Retain its
	// actual initial Kit artifact rather than rebuilding unrelated frontend files.
	if e = copyPublicationTree(filepath.Join(app, "web/build"), filepath.Join(dir, "web/build")); e != nil {
		t.Fatal(e)
	}
	replacement := startContractHost(t, ctx, dir, bin)
	defer replacement.stop()
	ready := replacement.mustCall(t, contractCommand{Op: "request", Method: "GET", Path: "/orders", Headers: map[string]string{"Accept": "text/html"}})
	if ready.Status != 200 || !strings.Contains(ready.Body, "<h1>Order total 9000</h1>") {
		t.Fatalf("rebuilt initialized literal9000 missing %+v", ready)
	}
	oldPID := old.cmd.Process.Pid
	e = build.Commit(func(*gen.CompilerInputs) error { return emit(build.Revision) })
	if e != nil {
		t.Fatal(e)
	}
	old.stop()
	if ready.PID == oldPID {
		t.Fatal("structural rebuild did not replace process")
	}
	t.Logf("REBUILD_PUBLISHED revision=%d oldPID=%d newPID=%d elapsed=%s literalHTTP=9000", build.Revision, oldPID, ready.PID, time.Since(started))
	// Keep this host's immutable compiler/dependency view and resume retained
	// compatible body edits on its stable paths, using subsequent Project leases.
	baseFiles := build.Inputs.Files()
	qNew := QualificationUse(t, "whole-retained")
	for _, n := range []int{9101, 9202} {
		body := strings.Replace(structural, "return total", fmt.Sprintf("return total*0+%d", n), 1)
		os.WriteFile(authored, []byte(body), 0644)
		if _, _, e = project.Observe(ctx, authored); e != nil {
			t.Fatal(e)
		}
		next, e := project.Prepare(ctx)
		if e != nil {
			t.Fatal(e)
		}
		nextFiles := next.Inputs.Files()
		frozen := map[string][]byte{}
		for path, b := range baseFiles {
			frozen[path] = b
		}
		for path, b := range nextFiles {
			if rel, e := filepath.Rel(next.Inputs.Dir, path); e == nil && rel != ".." && !strings.HasPrefix(rel, "../") && strings.HasSuffix(path, ".go") {
				frozen[filepath.Join(dir, rel)] = b
			}
		}
		qNew.frozen = frozen
		changed := map[string][]byte{}
		for path := range newBefore {
			changed[path] = frozen[path]
		}
		start := time.Now()
		candidate, e := newTC.CompileApplication(ctx, dir, pkgPath, cfgPath, filepath.Join(work, fmt.Sprintf("after-rebuild-%d", n)), newBefore, changed)
		if e != nil {
			t.Fatal(e)
		}
		prepared := time.Since(start)
		e = next.Commit(func(*gen.CompilerInputs) error {
			replacement.mustCall(t, contractCommand{Op: "install", Candidate: candidate})
			return emit(next.Revision)
		})
		if e != nil {
			t.Fatal(e)
		}
		next.Inputs.Close()
		r := replacement.mustCall(t, contractCommand{Op: "request", Method: "GET", Path: "/orders", Headers: map[string]string{"Accept": "text/html"}})
		want := fmt.Sprintf("<h1>Order total %d</h1>", n)
		if r.Status != 200 || !strings.Contains(r.Body, want) || r.PID != ready.PID {
			t.Fatalf("compatible edit after structural rebuild lost: %+v wanted%s", r, want)
		}
		t.Logf("POST_REBUILD_NATIVE revision=%d prepare=%s pid=%d literalHTTP=%d", next.Revision, prepared, r.PID, n)
	}
	checkEvents([]gen.Revision{1, 3, 4, 5, 6, 7, 8})
}
func copyPublicationTree(src, dst string) error {
	return filepath.WalkDir(src, func(path string, d os.DirEntry, e error) error {
		if e != nil {
			return e
		}
		rel, e := filepath.Rel(src, path)
		if e != nil {
			return e
		}
		target := filepath.Join(dst, rel)
		if d.IsDir() {
			return os.MkdirAll(target, 0755)
		}
		b, e := os.ReadFile(path)
		if e != nil {
			return e
		}
		return os.WriteFile(target, b, 0644)
	})
}
