//go:build skgo_retained_qualification

package hotreload

import (
	"crypto/sha256"
	"fmt"
	"github.com/tylergannon/polytype/goload"
	"go/parser"
	"go/token"
	"golang.org/x/mod/modfile"
	"golang.org/x/tools/go/packages"
	"os"
	"path/filepath"
	"sort"
	"strings"
	"testing"
)

// QualificationUse leaves CompileApplication's classifier/compiler/artifact/host
// path intact. Only acquisition differs. Each consumer owns its private app.
func QualificationUse(t *testing.T, mode string) *qualificationLoader {
	q := &qualificationLoader{t: t, mode: mode}
	q.session.SetMaxUniverses(4)
	retainedQualificationLoad = q.Load
	t.Cleanup(func() {
		if q.view != nil {
			q.view.Close()
		}
		retainedQualificationLoad = packages.Load
	})
	return q
}

type qualificationLoader struct {
	frozen          map[string][]byte
	t               *testing.T
	mode            string
	session         goload.Session
	view            *goload.View
	selection       string
	patterns        []string
	dependencyFiles map[string][]byte
	dependencyDirs  map[string]bool
	embeddedFiles   map[string][]byte
	initialized     bool
}

func (q *qualificationLoader) Load(c *packages.Config, patterns ...string) ([]*packages.Package, error) {
	if q.mode == "baseline" {
		return packages.Load(c, patterns...)
	}
	files := map[string][]byte{}
	var e error
	if q.frozen != nil {
		for p, b := range q.frozen {
			files[p] = b
		}
	} else {
		files, e = qualificationCapture(c.Dir)
	}
	if e != nil {
		return nil, e
	}
	for p, b := range c.Overlay {
		files[p] = b
	}
	if q.view == nil {
		q.view = q.session.PrepareView(files)
	} else if e = q.view.Update(files); e != nil {
		return nil, e
	}
	if q.mode == "exact-view" {
		return q.view.Load(c, patterns...)
	}
	if q.mode != "whole-retained" {
		return nil, fmt.Errorf("unknown acquisition mode %s", q.mode)
	}
	key := qualificationSelection(files)
	if key != q.selection {
		q.patterns = nil
		q.selection = key
	}
	meta := *c
	meta.Overlay = nil
	meta.Mode = packages.NeedName | packages.NeedFiles | packages.NeedCompiledGoFiles | packages.NeedImports | packages.NeedDeps | packages.NeedModule | packages.NeedEmbedFiles | packages.NeedTypesSizes
	query := q.patterns
	if len(query) == 0 {
		query = patterns
	}
	roots, e := q.view.Load(&meta, query...)
	if e != nil {
		return nil, e
	}
	all := map[string]*packages.Package{}
	packages.Visit(roots, nil, func(p *packages.Package) { all[p.PkgPath] = p })
	target := patterns[0]
	affected := map[string]bool{target: true}
	for changed := true; changed; {
		changed = false
		for path, p := range all {
			if affected[path] {
				continue
			}
			for dep := range p.Imports {
				if affected[dep] {
					affected[path] = true
					changed = true
					break
				}
			}
		}
	}
	selected := map[string]bool{}
	q.patterns = nil
	for _, p := range roots {
		selected[p.PkgPath] = true
		q.patterns = append(q.patterns, p.PkgPath)
	}
	// All application roots remain type checked; all reverse-import callers,
	// including local replacements outside ./..., become explicit source roots.
	for path := range affected {
		selected[path] = true
	}
	var check []string
	for path := range selected {
		check = append(check, path)
	}
	sort.Strings(check)
	if !q.initialized {
		q.dependencyFiles = map[string][]byte{}
		q.dependencyDirs = map[string]bool{}
		q.embeddedFiles = map[string][]byte{}
		for _, p := range all {
			for _, name := range p.EmbedFiles {
				b, e := os.ReadFile(name)
				if e != nil {
					return nil, e
				}
				q.embeddedFiles[name] = b
			}
		}
		var visitDeps func(*packages.Package)
		seen := map[string]bool{}
		visitDeps = func(p *packages.Package) {
			if p == nil || seen[p.PkgPath] {
				return
			}
			seen[p.PkgPath] = true
			for _, dep := range p.Imports {
				for _, name := range dep.CompiledGoFiles {
					q.dependencyDirs[filepath.Dir(name)] = true
					if b, ok := files[name]; ok {
						q.dependencyFiles[name] = b
					}
				}
				visitDeps(dep)
			}
		}
		visitDeps(all[target])
		for name, b := range files {
			if q.dependencyDirs[filepath.Dir(name)] {
				q.dependencyFiles[name] = b
			}
		}
		q.initialized = true
	} else {
		for _, p := range all {
			for _, name := range p.EmbedFiles {
				if _, ok := q.embeddedFiles[name]; !ok {
					return nil, &CompleteBuild{Reason: "embedded input membership changed; rebuild host: " + name}
				}
			}
		}
		for name, before := range q.embeddedFiles {
			now, e := os.ReadFile(name)
			if e != nil || string(now) != string(before) {
				return nil, &CompleteBuild{Reason: "embedded dependency input changed; rebuild host: " + name}
			}
		}
		for name := range files {
			if q.dependencyDirs[filepath.Dir(name)] {
				if _, ok := q.dependencyFiles[name]; !ok {
					return nil, &CompleteBuild{Reason: "dependency source membership changed; rebuild host: " + name}
				}
			}
		}
		for name, before := range q.dependencyFiles {
			if now, ok := files[name]; !ok || string(now) != string(before) {
				return nil, &CompleteBuild{Reason: "captured target dependency changed; refresh archives and rebuild host: " + name}
			}
		}
	}
	typed := *c
	typed.Mode &^= packages.NeedDeps
	typed.Mode |= packages.NeedEmbedFiles
	typed.Overlay = nil
	got, e := q.view.Load(&typed, check...)
	if e != nil {
		return nil, e
	}
	coverage := map[string]*packages.Package{}
	packages.Visit(got, nil, func(p *packages.Package) { coverage[p.PkgPath] = p })
	for path := range affected {
		p := coverage[path]
		if p == nil || p.TypesInfo == nil || len(p.Syntax) == 0 {
			return nil, fmt.Errorf("missing complete caller coverage: %s", path)
		}
	}
	q.t.Logf("WHOLE retained applicationRoots=%d typedRoots=%d graphPackages=%d affected=%d stats=%+v", len(roots), len(got), len(coverage), len(affected), q.session.Stats())
	return got, nil
}
func qualificationCapture(root string) (map[string][]byte, error) {
	files := map[string][]byte{}
	seen := map[string]bool{}
	var capture func(string) error
	capture = func(dir string) error {
		dir, _ = filepath.Abs(dir)
		if seen[dir] {
			return nil
		}
		seen[dir] = true
		err := filepath.WalkDir(dir, func(path string, d os.DirEntry, e error) error {
			if e != nil {
				return e
			}
			if d.IsDir() {
				if path != dir {
					switch d.Name() {
					case ".git", "node_modules", ".svelte-kit", ".skgo", "ephemeral", "docs":
						return filepath.SkipDir
					}
					if _, e := os.Stat(filepath.Join(path, "go.mod")); e == nil {
						return filepath.SkipDir
					}
				}
				return nil
			}
			name := d.Name()
			if name != "" {
				b, e := os.ReadFile(path)
				if e != nil {
					return e
				}
				files[path] = b
			}
			return nil
		})
		if err != nil {
			return err
		}
		raw, e := os.ReadFile(filepath.Join(dir, "go.mod"))
		if e != nil {
			return e
		}
		m, e := modfile.Parse("go.mod", raw, nil)
		if e != nil {
			return e
		}
		for _, r := range m.Replace {
			if r.New.Version == "" {
				replacement := r.New.Path
				if !filepath.IsAbs(replacement) {
					replacement = filepath.Join(dir, replacement)
				}
				if e := capture(replacement); e != nil {
					return e
				}
			}
		}
		return nil
	}
	return files, capture(root)
}
func qualificationSelection(files map[string][]byte) string {
	var names []string
	for p := range files {
		names = append(names, p)
	}
	sort.Strings(names)
	h := sha256.New()
	for _, p := range names {
		fmt.Fprintln(h, p)
		b := files[p]
		if strings.HasSuffix(p, ".go") {
			f, e := parser.ParseFile(token.NewFileSet(), p, b, parser.ImportsOnly|parser.ParseComments)
			if e != nil {
				h.Write(b)
				continue
			}
			fmt.Fprintln(h, f.Name.Name)
			for _, im := range f.Imports {
				fmt.Fprintln(h, im.Path.Value)
			}
			for _, g := range f.Comments {
				for _, c := range g.List {
					if strings.HasPrefix(c.Text, "//go:") {
						fmt.Fprintln(h, c.Text)
					}
				}
			}
		} else {
			h.Write(b)
		}
	}
	return fmt.Sprintf("%x", h.Sum(nil))
}
