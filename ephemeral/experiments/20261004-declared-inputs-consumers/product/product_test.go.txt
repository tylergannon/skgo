func put(t *testing.T, root, name, body string) {
	t.Helper()

func read(t *testing.T, root, name string) string {

func exact(t *testing.T, got, want string) {
	t.Helper()

}
func copyTree(t *testing.T, source, dest string) {
	t.Helper()
	e := filepath.WalkDir(source, func(path string, d fs.DirEntry, e error) error {
		if e != nil {
			return e
		}
		rel, e := filepath.Rel(source, path)
		if e != nil {
			return e
		}
		if rel == "web/node_modules" || rel == "web/.svelte-kit" || rel == "web/build" || rel == "web/dist" {
			return filepath.SkipDir
		}
		target := filepath.Join(dest, rel)
		if d.IsDir() {
			return os.MkdirAll(target, 0755)
		}
		info, e := d.Info()
		if e != nil {
			return e
		}
		if info.Mode()&os.ModeSymlink != 0 {
			link, e := os.Readlink(path)
			if e != nil {
				return e
			}
			return os.Symlink(link, target)
		}
		b, e := os.ReadFile(path)
		if e != nil {
			return e
		}
		return os.WriteFile(target, b, info.Mode().Perm())
	})
	if e != nil {
		t.Fatal(e)
	}
}

func run(t *testing.T, root, log string, env []string, args ...string) (string, error) {
	t.Helper()
	ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
	defer cancel()
	cmd := exec.CommandContext(ctx, args[0], args[1:]...)
	cmd.Dir = root
	cmd.Env = append(os.Environ(), append([]string{"GO111MODULE=on", "GOWORK=off", "GOCACHE=/tmp/skgo-inputs-proof-go-cache", "MISE_STATE_DIR=" + base + "/mise-state", "MISE_CACHE_DIR=" + base + "/mise-cache", "MISE_TRUSTED_CONFIG_PATHS=" + base, "MISE_YES=1"}, env...)...)
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	var out bytes.Buffer
	cmd.Stdout = &out
	cmd.Stderr = &out
	if e := cmd.Start(); e != nil {
		t.Fatal(e)
	}
	t.Logf("command pid/group=%d cwd=%s command=%v", cmd.Process.Pid, root, args)
	done := make(chan error, 1)
	go func() { done <- cmd.Wait() }()
	var e error
	select {
	case e = <-done:
	case <-ctx.Done():
		_ = syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL)
		e = <-done
		t.Errorf("command exceeded 60s deadline")
	}
	put(t, root, log, out.String())
	return out.String(), e
}
func generate(t *testing.T, root string, env []string) {

		t.Fatalf("generate: %v\n%s", e, b)
	}
}

func cleanupRecorded(t *testing.T, ready string) {
	t.Helper()
	b, e := os.ReadFile(ready)
	if e != nil {
		return
	}
	var pid int
	var dir string
	if _, e = fmt.Sscanf(string(b), "%d %s", &pid, &dir); e == nil && pid > 0 {
		_ = syscall.Kill(-pid, syscall.SIGKILL)
		if strings.HasPrefix(dir, os.TempDir()) || strings.Contains(dir, "/skgo-prerender-") {
			_ = os.RemoveAll(dir)
		}
	}
}

func TestIndependentSIGTERMJoinsBlockedOwnedBody(t *testing.T) {

	mainPID := filepath.Join(root, "main.pid")

	generate(t, root, env)
	config := read(t, app, "web/vite.config.ts")
	config = `import {isMainThread,threadId} from 'node:worker_threads';import {writeFileSync,realpathSync} from 'node:fs';import {createRequire} from 'node:module';import {dirname,join} from 'node:path';const ownRequire=createRequire(process.cwd()+'/package.json');const ownVite=ownRequire.resolve('vite');writeFileSync(process.env.SKGO_MAIN_PID!,String(process.pid));const ownerSelfReceipt=()=>{if(isMainThread)writeFileSync(process.env.SKGO_MAIN_PID!+'.info',JSON.stringify({isMainThread,threadId,pid:process.pid,ppid:process.ppid,execPath:process.execPath,argv:process.argv,packageManager:process.env.NODE_PACKAGE_MANAGER??null,vite:ownVite,viteCLI:realpathSync(join(dirname(ownVite),'cli.js'))}));};if(isMainThread){ownerSelfReceipt();setInterval(ownerSelfReceipt,50).unref();}` + "\n" + config
	put(t, app, "web/vite.config.ts", config)
	cmd := exec.Command("just", "build")
	cmd.Dir = root
	if os.Getenv("SKGO_SIGNAL_OUTER") == "1" {
		cmd = exec.Command(filepath.Join(app, "web/node_modules/.bin/vp"), "build")
		cmd.Dir = filepath.Join(app, "web")
	}
	cmd.Env = append(os.Environ(), append([]string{"GO111MODULE=on", "GOWORK=off", "GOCACHE=/tmp/skgo-inputs-proof-go-cache", "MISE_STATE_DIR=" + base + "/mise-state", "MISE_CACHE_DIR=" + base + "/mise-cache", "MISE_TRUSTED_CONFIG_PATHS=" + base, "MISE_YES=1"}, env...)...)
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	var out bytes.Buffer
	cmd.Stdout = &out
	cmd.Stderr = &out
	if e := cmd.Start(); e != nil {
		t.Fatal(e)
	}
	defer func() { _ = syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL); cleanupRecorded(t, ready) }()
	done := make(chan error, 1)
	go func() { done <- cmd.Wait() }()
	deadline := time.Now().Add(35 * time.Second)
	for {
		if _, e := os.Stat(ready); e == nil {
			break
		}
		if time.Now().After(deadline) {
			t.Fatal("actual blocked Go body not ready before signal deadline")
		}
		select {
		case e := <-done:
			t.Fatalf("build ended before blocked readiness: %v\n%s", e, out.String())
		default:
		}
		time.Sleep(15 * time.Millisecond)
	}
	var nativePID int
	if _, e := fmt.Sscanf(read(t, root, "main.pid"), "%d", &nativePID); e != nil {
		t.Fatal(e)
	}
	var recordedBodyPID int
	var recordedDir string
	fmt.Sscanf(read(t, root, "body.ready"), "%d %s", &recordedBodyPID, &recordedDir)
	ps := exec.Command("ps", "-o", "pid=,ppid=,pgid=,stat=,comm=", "-p", fmt.Sprintf("%d,%d,%d", cmd.Process.Pid, nativePID, recordedBodyPID))
	tree, _ := ps.CombinedOutput()
	put(t, root, "tree-before.log", string(tree))
	put(t, root, "node-before.json", read(t, root, "main.pid.info"))
	signalPID := nativePID
	if os.Getenv("SKGO_SIGNAL_OUTER") == "1" {
		signalPID = cmd.Process.Pid
	}
	t.Logf("signal target%d outer%d nativeNode%d blockedGo%d", signalPID, cmd.Process.Pid, nativePID, recordedBodyPID)
	if e := syscall.Kill(signalPID, syscall.SIGTERM); e != nil {
		t.Fatal(e)
	}
	select {
	case e := <-done:
		if e == nil {
			t.Fatal("SIGTERM build returned success")
		}
	case <-time.After(20 * time.Second):
		ps = exec.Command("ps", "-o", "pid=,ppid=,pgid=,stat=,comm=", "-p", fmt.Sprintf("%d,%d,%d", cmd.Process.Pid, nativePID, recordedBodyPID))
		tree, _ = ps.CombinedOutput()
		put(t, root, "tree-after-timeout.log", string(tree))
		put(t, root, "build.log", out.String())
		put(t, root, "node-after.json", read(t, root, "main.pid.info"))
		outerAlive := syscall.Kill(cmd.Process.Pid, 0)
		t.Logf("outer alive probe after TERM: %v; native before%s after%s; Go%s", outerAlive, read(t, root, "node-before.json"), read(t, root, "node-after.json"), read(t, root, "body.ready"))
		t.Fatal("SIGTERM target did not join within20s deadline")
	}
	put(t, root, "build.log", out.String())
	var bodyPID int
	var dir string
	if _, e := fmt.Sscanf(read(t, root, "body.ready"), "%d %s", &bodyPID, &dir); e != nil {
		t.Fatal(e)
	}
	if e := syscall.Kill(-bodyPID, 0); e != syscall.ESRCH {
		t.Fatalf("owned Go group alive after main signal exit: pid%d err%v", bodyPID, e)
	}
	if _, e := os.Stat(dir); !os.IsNotExist(e) {
		t.Fatalf("private compile dir exists after main exit: %s %v", dir, e)
	}
	before := read(t, root, "late.log")
	time.Sleep(250 * time.Millisecond)
	exact(t, read(t, root, "late.log"), before)
	t.Logf("SIGTERM main PID%d joined body group%d; owned children0, tempdirgone, lateIO0", nativePID, bodyPID)
}
