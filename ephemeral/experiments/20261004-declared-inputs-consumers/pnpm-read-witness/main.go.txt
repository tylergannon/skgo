func must(err error) { if err != nil { panic(err) } }
func tree(root string) map[string]string {
 result:=map[string]string{}
 must(filepath.WalkDir(root,func(path string,d fs.DirEntry,err error)error{
  if err!=nil{return err}; rel,err:=filepath.Rel(root,path);if err!=nil{return err}
  if d.IsDir(){result[rel]="dir";return nil}; b,err:=os.ReadFile(path);if err!=nil{return err};result[rel]=fmt.Sprintf("%x",sha256.Sum256(b));return nil
 }));return result
}
func run(root,label string,extra []string,args ...string) string {
 cmd:=exec.Command("/Users/tyler/.vite-plus/bin/pnpm",args...);cmd.Dir=root
 for _,value:=range os.Environ(){if !strings.HasPrefix(value,"PNPM_CONFIG_PATCHED_DEPENDENCIES="){cmd.Env=append(cmd.Env,value)}}
 cmd.Env=append(cmd.Env,extra...)
 out,err:=cmd.CombinedOutput();fmt.Printf("%s args=%q exit=%v output=%s\n",label,args,err,out);must(err);return strings.TrimSpace(string(out))
}
