package renderer

import (
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net"
	"net/http"
	"net/http/httptest"
	"net/url"
	"os"
	"os/exec"
	"path/filepath"
	"runtime"
	"strings"
	"sync"
	"sync/atomic"
	"syscall"
	"testing"
	"time"

	"github.com/tylergannon/skgo/internal/ssr"
)

func TestRendererArchitecture(t *testing.T) {
	root, err := filepath.Abs("../../../..")
	if err != nil {
		t.Fatal(err)
	}
	dir, err := os.MkdirTemp(".", "fixture-")
	if err != nil {
		t.Fatal(err)
	}
	dir, _ = filepath.Abs(dir)
	defer func() {
		raw, _ := os.ReadFile(filepath.Join(dir, "vite.log"))
		_ = os.WriteFile("vite-last.log", raw, 0644)
		if os.Getenv("SKGO_KEEP_RENDERER_FIXTURE") == "1" {
			t.Log("retained fixture", dir)
		} else {
			os.RemoveAll(dir)
		}
	}()
	write := func(name, body string) {
		t.Helper()
		filename := filepath.Join(dir, name)
		if err := os.MkdirAll(filepath.Dir(filename), 0755); err != nil {
			t.Fatal(err)
		}
		if err := os.WriteFile(filename, []byte(body), 0644); err != nil {
			t.Fatal(err)
		}
	}
	manifest, err := os.ReadFile(filepath.Join(root, "example/web/package.json"))
	if err != nil {
		t.Fatal(err)
	}
	write("package.json", string(manifest))
	if err := os.Symlink(filepath.Join(root, "example/web/node_modules"), filepath.Join(dir, "node_modules")); err != nil {
		t.Fatal(err)
	}
	config, err := os.ReadFile("vite.config.js")
	if err != nil {
		t.Fatal(err)
	}
	write("vite.config.js", strings.NewReplacer("ADAPTER_ENV", filepath.Join(root, "internal/adapter/skgo-adapter/env.js"), "POLYFILL", filepath.Join(root, "internal/adapter/skgo-adapter/polyfill.js")).Replace(string(config)))
	write("src/app.html", `<!doctype html><html><head>%sveltekit.head%</head><body><div>%sveltekit.body%</div></body></html>`)
	write("src/routes/+layout.svelte", `<script>let { children } = $props();</script><main>{@render children()}</main>`)
	write("src/routes/+page.svelte", `<script>import {label} from './message.js';</script><h1>Component A: {label}</h1>`)
	write("src/routes/message.js", `export const label = 'Dependency A';`)
	write("runtime-entry.js", `import {ModuleRunner} from 'vite/module-runner';
 const NativeURL = globalThis.URL; globalThis.URL=class extends NativeURL {constructor(...args){if(typeof args[0]==='string' && args[0].startsWith('file:'))args[0]=args[0].replaceAll('\u0000','%00');try{super(...args)}catch(error){throw new Error('URL constructor inputs '+JSON.stringify(args)+': '+error.message)}}};
 const runner=new ModuleRunner({sourcemapInterceptor:false,hmr:false,transport:{invoke:payload=>__skgo_remote('module',JSON.stringify(payload)).then(JSON.parse)}});
 globalThis.__skgo_ping=()=> 'ok';
 globalThis.__skgo_render = req => {
 let result, failure='';
 const output={};for(const key of ['done','failure','body','head','status','error','redirect','fetched'])Object.defineProperty(output,key,{get:()=>key==='failure'?failure || result?.failure:key==='done'?!!failure || result?.done:result?.[key]});
 runner.import(`+fmt.Sprintf("%q", filepath.Join(root, "internal/adapter/skgo-adapter/entry.js"))+`).then(()=>{result=globalThis.__skgo_render(req)},error=>{failure=error.stack || String(error)});
 return output;
 };`)

	bootstrap, err := os.ReadFile(filepath.Join(dir, "runtime-entry.js"))
	if err != nil {
		t.Fatal(err)
	}
	retained := strings.Replace(string(bootstrap), "const runner=new ModuleRunner", "let receive, cursor=0; const runner=new ModuleRunner", 1)
	retained = strings.Replace(retained, "hmr:false,transport:{invoke:", "hmr:{logger:false},transport:{connect(handlers){receive=handlers.onMessage},invoke:", 1)
	retained = strings.Replace(retained, "runner.import(", "__skgo_remote('events',String(cursor)).then(JSON.parse).then(async batch=>{for(const payload of batch.events)void payload;cursor=batch.cursor;return runner.import(", 1)
	retained = strings.Replace(retained, ").then(()=>{result=globalThis.__skgo_render(req)}", ")}).then(()=>{result=globalThis.__skgo_render(req)}", 1)
	write("retained-entry.js", retained)

	listener, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	port := listener.Addr().(*net.TCPAddr).Port
	listener.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Minute)
	defer cancel()
	log, err := os.Create(filepath.Join(dir, "vite.log"))
	if err != nil {
		t.Fatal(err)
	}
	defer log.Close()
	cmd := exec.CommandContext(ctx, "vp", "dev", "--host", "127.0.0.1", "--port", fmt.Sprint(port), "--strictPort")
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	cmd.Dir = dir
	cmd.Stdout = log
	cmd.Stderr = log
	if err := cmd.Start(); err != nil {
		t.Fatal(err)
	}
	defer func() {
		syscall.Kill(-cmd.Process.Pid, syscall.SIGTERM)
		cancel()
		cmd.Wait()
		if t.Failed() {
			raw, _ := os.ReadFile(filepath.Join(dir, "vite.log"))
			t.Log(string(raw))
		}
	}()
	client := &http.Client{Timeout: 45 * time.Second}
	base := fmt.Sprintf("http://127.0.0.1:%d/__experiment", port)
	for deadline := time.Now().Add(20 * time.Second); ; {
		res, err := client.Get(base)
		if err == nil {
			res.Body.Close()
			break
		}
		if time.Now().After(deadline) {
			t.Fatal("Vite did not start:", err)
		}
		time.Sleep(100 * time.Millisecond)
	}
	call := func(operation string) time.Duration {
		t.Helper()
		start := time.Now()
		res, err := client.Get(base + operation)
		if err != nil {
			t.Fatal(err)
		}
		defer res.Body.Close()
		raw, _ := io.ReadAll(res.Body)
		if res.StatusCode != 200 {
			t.Fatalf("%s: %s", operation, raw)
		}
		t.Logf("%s wall=%s vite=%s", operation, time.Since(start), raw)
		return time.Since(start)
	}

	request, _ := json.Marshal(ssr.Request{URL: "http://localhost:8080/", RouteID: "/", Status: 200, Branch: []ssr.Node{{Index: 0}, {Index: 1}}})
	var requests, bytes atomic.Int64
	hosts := ssr.Hosts{Remote: func(ctx context.Context, id, payload string) ([]byte, error) {
		requests.Add(1)
		endpoint := base + "/fetch?" + url.QueryEscape(payload)
		if id == "events" {
			endpoint = base + "/events?" + payload
		}
		req, _ := http.NewRequestWithContext(ctx, "GET", endpoint, nil)
		response, err := client.Do(req)
		if err != nil {
			return nil, err
		}
		defer response.Body.Close()
		data, err := io.ReadAll(response.Body)
		bytes.Add(int64(len(data)))
		if response.StatusCode != 200 {
			return nil, fmt.Errorf("module fetch: %s", data)
		}
		return data, err
	}}
	newEngine := func(name string) *ssr.Engine {
		t.Helper()
		raw, err := os.ReadFile(filepath.Join(dir, name))
		if err != nil {
			t.Fatal(err)
		}
		start := time.Now()
		engine, err := ssr.New(name, raw, 2, nil)
		if err != nil {
			t.Fatal(err)
		}
		t.Logf("POOL name=%s artifact_bytes=%d prepare_us=%d", name, len(raw), time.Since(start).Microseconds())
		return engine
	}
	check := func(engine *ssr.Engine, want, stage string) {
		t.Helper()
		calls, transferred := requests.Load(), bytes.Load()
		start := time.Now()
		result, _, err := engine.Render(ctx, "/", request, hosts)
		if err != nil {
			t.Fatalf("%s BLOCKED: %v", stage, err)
		}
		if !strings.Contains(result.Body, want) {
			t.Fatalf("%s got %q want %q", stage, result.Body, want)
		}
		t.Logf("RENDER stage=%s elapsed_us=%d module_calls=%d wire_bytes=%d runtimes=%d expected=%q", stage, time.Since(start).Microseconds(), requests.Load()-calls, bytes.Load()-transferred, engine.Created(), want)
	}
	httpCheck := func(engine *ssr.Engine, want, stage string) {
		t.Helper()
		server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			result, _, err := engine.Render(r.Context(), "/", request, hosts)
			if err != nil {
				http.Error(w, err.Error(), 500)
				return
			}
			w.Header().Set("Content-Type", "text/html")
			io.WriteString(w, result.Body)
		}))
		defer server.Close()
		var wg sync.WaitGroup
		start := make(chan struct{})
		errs := make(chan string, 8)
		for i := 0; i < 8; i++ {
			wg.Add(1)
			go func() {
				defer wg.Done()
				<-start
				res, err := client.Get(server.URL)
				if err != nil {
					errs <- err.Error()
					return
				}
				defer res.Body.Close()
				body, _ := io.ReadAll(res.Body)
				if res.StatusCode != 200 || !strings.Contains(string(body), want) {
					errs <- fmt.Sprintf("status %d body %s", res.StatusCode, body)
				}
			}()
		}
		close(start)
		wg.Wait()
		close(errs)
		for err := range errs {
			t.Error(stage, err)
		}
		limit := 2
		// The rejected render deliberately discards one VM; Created counts
		// lifetime creation, not current live slots (ssr.go Render cleanup).
		if stage == "retained-repair-concurrent" {
			limit = 3
		}
		if engine.Created() > limit {
			t.Fatalf("unexpected runtime creation: got %d, fixture limit %d", engine.Created(), limit)
		}
		t.Logf("HTTP stage=%s requests=8 pool_created=%d pool_size=%d literal=%q", stage, engine.Created(), engine.Size(), want)
	}
	seen := func() map[string]int {
		res, err := client.Get(base + "/seen")
		if err != nil {
			t.Fatal(err)
		}
		defer res.Body.Close()
		var values map[string]int
		if err = json.NewDecoder(res.Body).Decode(&values); err != nil {
			t.Fatal(err)
		}
		return values
	}
	edit := func(file, body string) {
		t.Helper()
		filename := filepath.Join(dir, file)
		before := seen()[filename]
		write(file, body)
		for deadline := time.Now().Add(5 * time.Second); ; {
			if seen()[filename] > before {
				break
			}
			if time.Now().After(deadline) {
				t.Fatalf("Vite watcher did not observe %s", file)
			}
			time.Sleep(10 * time.Millisecond)
		}
	}
	call("/full")
	artifact := newEngine("bundle.js")
	check(artifact, "Component A: Dependency A", "artifact-initial")
	call("/runtime")
	live := newEngine("runtime.js")
	check(live, "Component A: Dependency A", "live-initial-cold")
	check(live, "Component A: Dependency A", "live-initial-warm")
	httpCheck(live, "Component A: Dependency A", "live-initial-concurrent")
	httpCheck(artifact, "Component A: Dependency A", "artifact-initial-concurrent")
	retainedEngine := newEngine("retained.js")
	check(retainedEngine, "Component A: Dependency A", "retained-initial")
	httpCheck(retainedEngine, "Component A: Dependency A", "retained-initial-concurrent")
	for i := 1; i <= 4; i++ {
		label := fmt.Sprintf("Dependency %d", i)
		edit("src/routes/message.js", fmt.Sprintf("export const label = %q;", label))
		check(retainedEngine, "Component A: "+label, fmt.Sprintf("retained-dependency-%d", i))
		httpCheck(retainedEngine, "Component A: "+label, fmt.Sprintf("retained-dependency-%d-concurrent", i))
		call("/update")
		artifact = newEngine("bundle.js")
		check(artifact, "Component A: "+label, fmt.Sprintf("artifact-dependency-%d", i))
		live = newEngine("runtime.js")
		check(live, "Component A: "+label, fmt.Sprintf("live-dependency-%d", i))
		check(live, "Component A: "+label, fmt.Sprintf("live-warm-%d", i))
	}
	// Add a component and change the page's import edge without replacing the
	// dev server or its Vite-owned graph. This deliberately does not claim new routes.
	write("src/routes/extra.svelte", `<section>New imported component: {23 * 3}</section>`)
	edit("src/routes/+page.svelte", `<script>import {label} from './message.js';import Extra from './extra.svelte';</script><h1>Component B: {label}</h1><Extra />`)
	check(retainedEngine, "New imported component: 69", "retained-import-edge")
	call("/update")
	artifact = newEngine("bundle.js")
	check(artifact, "New imported component: 69", "artifact-import-edge")
	live = newEngine("runtime.js")
	check(live, "New imported component: 69", "live-import-edge")
	httpCheck(live, "Component B: Dependency 4", "live-import-edge-concurrent")
	httpCheck(artifact, "Component B: Dependency 4", "artifact-import-edge-concurrent")
	// An invalid source must fail candidate preparation. Preserve the previously
	// installed artifact engine; a candidate pool is admitted only after rendering.
	edit("src/routes/message.js", `export const label = ;`)
	res, err := client.Get(base + "/update")
	if err != nil {
		t.Fatal(err)
	}
	body, _ := io.ReadAll(res.Body)
	res.Body.Close()
	if res.StatusCode != 500 {
		t.Fatalf("invalid artifact accepted: %d %s", res.StatusCode, body)
	}
	t.Logf("EXPECTED_FAILURE artifact status=%d diagnostic=%s", res.StatusCode, body)
	check(artifact, "Component B: Dependency 4", "artifact-last-good-after-error")
	candidate := newEngine("runtime.js")
	_, _, err = candidate.Render(ctx, "/", request, hosts)
	if err == nil {
		t.Fatal("invalid live candidate unexpectedly rendered")
	}
	t.Logf("EXPECTED_FAILURE live %v", err)
	retainedResult, _, retainedErr := retainedEngine.Render(ctx, "/", request, hosts)
	if retainedErr == nil && !strings.Contains(retainedResult.Body, "Component B: Dependency 4") {
		t.Fatalf("retained error produced neither useful failure nor last-good value: %q", retainedResult.Body)
	}
	t.Logf("RETAINED_HMR_AFTER_ERROR err=%v body=%q", retainedErr, retainedResult.Body)
	// A retained upstream runner with HMR disabled is observed, not asserted as
	// a supported update/lifetime policy. Warm module caches may mask bad source.
	oldResult, _, oldErr := live.Render(ctx, "/", request, hosts)
	t.Logf("RETAINED_LIVE_AFTER_ERROR err=%v body=%q", oldErr, oldResult.Body)
	edit("src/routes/message.js", `export const label = 'Repaired 901';`)
	call("/update")
	artifact = newEngine("bundle.js")
	check(artifact, "Component B: Repaired 901", "artifact-repair")
	live = newEngine("runtime.js")
	check(live, "Component B: Repaired 901", "live-repair")
	httpCheck(live, "Component B: Repaired 901", "live-repair-concurrent")
	httpCheck(artifact, "Component B: Repaired 901", "artifact-repair-concurrent")
	oldResult, _, oldErr = retainedEngine.Render(ctx, "/", request, hosts)
	if oldErr != nil || !strings.Contains(oldResult.Body, "Component B: Repaired 901") {
		t.Fatalf("retained repair err=%v body=%q", oldErr, oldResult.Body)
	}
	httpCheck(retainedEngine, "Component B: Repaired 901", "retained-repair-concurrent")
	runtime.GC()
	var memory runtime.MemStats
	runtime.ReadMemStats(&memory)
	nodeMemory, memoryErr := client.Get(base + "/resources")
	if memoryErr != nil {
		t.Fatal(memoryErr)
	}
	nodeBytes, _ := io.ReadAll(nodeMemory.Body)
	nodeMemory.Body.Close()
	t.Logf("NODE_RESOURCE %s", nodeBytes)
	runtime.KeepAlive(artifact)
	runtime.KeepAlive(live)
	runtime.KeepAlive(retainedEngine)
	t.Logf("RESOURCE final_go_heap_bytes=%d heap_objects=%d goroutines=%d module_calls_total=%d module_bytes_total=%d (single observation, not bounded-retention proof)", memory.HeapAlloc, memory.HeapObjects, runtime.NumGoroutine(), requests.Load(), bytes.Load())
}
