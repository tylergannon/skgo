//go:build skgo_retained_qualification

package hotreload

import (
	"context"
	"fmt"
	"io"
	"net/http"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"syscall"
	"testing"
	"time"
)

func qualificationBrowser(t *testing.T, ctx context.Context, cancel context.CancelFunc, tc Toolchain, app, pkgPath, source string, compile func(string, string, string) (Candidate, error), process *contractProcess) {
	address := process.mustCall(t, contractCommand{Op: "address"}).Body
	web := filepath.Join(app, "web")
	config := `export default {cacheDir:'.qualification-vite',server:{host:'127.0.0.1',port:56511,strictPort:true,proxy:{'^/(?!@|__tool/)':{target:'BACKEND',changeOrigin:false}}},plugins:[{name:'qualification-events',configureServer(server){server.middlewares.use('/__tool',async(req,res)=>{if(req.url.startsWith('/notify')){const error=new URL(req.url,'http://localhost').searchParams.get('error');server.hot.send({type:'custom',event:'qualification:ready',data:{error:error||''}});}res.end('ok')});}}]}`
	if e := os.WriteFile(filepath.Join(web, "vite.config.ts"), []byte(strings.ReplaceAll(config, "BACKEND", address)), 0644); e != nil {
		t.Fatal(e)
	}
	log, e := os.Create(filepath.Join(app, "vite-browser.log"))
	if e != nil {
		t.Fatal(e)
	}
	defer log.Close()
	v := exec.CommandContext(ctx, "vp", "dev")
	v.Dir = web
	v.Stdout = log
	v.Stderr = log
	v.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	if e = v.Start(); e != nil {
		t.Fatal(e)
	}
	done := make(chan error, 1)
	go func() { done <- v.Wait() }()
	defer func() { syscall.Kill(-v.Process.Pid, syscall.SIGKILL); <-done }()
	client := http.Client{Timeout: 10 * time.Second}
	notify := func(message string) {
		res, e := client.Get("http://127.0.0.1:56511/__tool/notify" + message)
		if e != nil {
			t.Logf("NOTIFY_FAILED %v", e)
			return
		}
		io.Copy(io.Discard, res.Body)
		res.Body.Close()
		if res.StatusCode != 200 {
			t.Logf("NOTIFY_FAILED status=%d", res.StatusCode)
		}
	}
	for end := time.Now().Add(20 * time.Second); ; {
		res, e := client.Get("http://127.0.0.1:56511/__tool/identity")
		if e == nil {
			res.Body.Close()
			break
		}
		if time.Now().After(end) {
			t.Fatal(e)
		}
		time.Sleep(100 * time.Millisecond)
	}
	sourceFile := filepath.Join(app, "web/src/routes/orders/orders.remote.go")
	last := source
	revision := 0
	t.Logf("BROWSER_READY public=http://127.0.0.1:56511/orders app=%s source=%s nativePID=%d mode=whole-retained wholeApplication=true HostView=true", app, sourceFile, process.cmd.Process.Pid)
	for {
		select {
		case <-ctx.Done():
			return
		case <-time.After(75 * time.Millisecond):
		}
		if _, e := os.Stat(filepath.Join(app, "stop-browser")); e == nil {
			cancel()
			return
		}
		b, e := os.ReadFile(sourceFile)
		if e != nil {
			t.Fatal(e)
		}
		if string(b) == last {
			continue
		}
		last = string(b)
		revision++
		start := time.Now()
		retainedQualificationStage = func(stage string, d time.Duration) {
			t.Logf("BROWSER_STAGE revision=%d stage=%s elapsed=%s", revision, stage, d)
		}
		candidate, e := compile(fmt.Sprintf("browser-%d", revision), string(b), contractPage)
		prepared := time.Since(start)
		if e != nil {
			t.Logf("BROWSER_REJECT revision=%d elapsed=%s diagnostic=%v", revision, prepared, e)
			notify("?error=GO_COMPILE_REJECTED")
			continue
		}
		// A later source save supersedes this candidate before it can enter the host.
		latest, e := os.ReadFile(sourceFile)
		if e != nil {
			t.Fatal(e)
		}
		if string(latest) != last {
			t.Logf("BROWSER_OBSOLETE revision=%d", revision)
			continue
		}
		start = time.Now()
		ack := process.mustCall(t, contractCommand{Op: "install", Candidate: candidate})
		installed := time.Since(start)
		res := process.mustCall(t, contractCommand{Op: "request", Method: "GET", Path: "/orders", Headers: map[string]string{"Accept": "text/html"}})
		if res.Status != 200 || !strings.Contains(res.Body, "Order total") {
			t.Fatalf("render after install failed: %+v", res)
		}
		t.Logf("BROWSER_NATIVE revision=%d prepare=%s install=%s pid=%d", revision, prepared, installed, ack.PID)
		notify("")
	}
}
